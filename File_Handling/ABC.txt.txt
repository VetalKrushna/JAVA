
CREATE TABLE Bus (bus_no INT PRIMARY KEY, capacity INT NOT NULL,depot_name VARCHAR(50) NOT NULL);

CREATE TABLE Driver (driver_no INT PRIMARY KEY,driver_name VARCHAR(50) NOT NULL,
  license_no VARCHAR(20) NOT NULL,address VARCHAR(100) NOT NULL,
  age INT NOT NULL CHECK (age > 18),salary int NOT NULL);

CREATE TABLE Bus_Driver (bus_no INT NOT NULL  REFERENCES Bus (bus_no),
  driver_no INT NOT NULL REFERENCES Driver (driver_no),
  date_of_duty DATE NOT NULL);

C Write a query to find the names of all drivers who were driving buses 
from “Kothrud Depot” on 12/2/2009. 
=SELECT d.driver_name
FROM Bus b
JOIN Bus_Driver bd
ON b.bus_no = bd.bus_no
JOIN Driver d
ON bd.driver_no = d.driver_no
WHERE b.depot_name = 'Kothrud Depot'
AND bd.date_of_duty = '2009-12-02';

Create a View to find all details of buses which are driven by drivers
whose age is less than 25 years

= create view bd as SELECT b.*, d.*
FROM Bus b
JOIN Bus_Driver bd
ON b.bus_no = bd.bus_no
JOIN Driver d
ON bd.driver_no = d.driver_no
WHERE d.age < 25;
